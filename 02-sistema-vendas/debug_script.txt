    dispararImpressao();
  } else {
    pw.addEventListener("load", dispararImpressao);
  }

  // Fecha a janela somente apos a conclusao (ou cancelamento) do dialogo
  // de impressao, disparado pelo evento nativo "afterprint". Um timeout

}
